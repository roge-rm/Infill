cmake_minimum_required(VERSION 3.22)
project(infill_desktop CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# A cross build passes JNI_INCLUDE_DIRS itself.
if(NOT JNI_INCLUDE_DIRS)
    find_package(JNI REQUIRED)
endif()
find_package(Threads REQUIRED)
set(CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp")
add_library(infill_audio SHARED engine.cpp music.cpp ${CPP}/synth/synth.cpp)
target_include_directories(infill_audio PRIVATE ${CPP} miniaudio ${JNI_INCLUDE_DIRS})
target_link_libraries(infill_audio PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
