cmake_minimum_required(VERSION 3.22)
project(infill_desktop CXX)
if(WIN32)
    enable_language(C RC)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# A cross build passes JNI_INCLUDE_DIRS itself.
if(NOT JNI_INCLUDE_DIRS)
    find_package(JNI REQUIRED)
endif()
find_package(Threads REQUIRED)
set(CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp")
add_library(infill_audio SHARED engine.cpp music.cpp ${CPP}/synth/synth.cpp)
target_include_directories(infill_audio PRIVATE ${CPP} miniaudio ${JNI_INCLUDE_DIRS})
if(WIN32)
    # miniaudio uses WASAPI here. Linked static so the DLL needs nothing but Windows.
    target_link_libraries(infill_audio PRIVATE ole32)
    target_link_options(infill_audio PRIVATE -static -static-libgcc -static-libstdc++)
    set_target_properties(infill_audio PROPERTIES PREFIX "")

    # Infill.exe starts the app on the Java runtime beside it (windows/launcher.c).
    set(INFILL_VERSION "0.0" CACHE STRING "The version the launcher reports")
    set(LAUNCHER_ICON ${CMAKE_CURRENT_SOURCE_DIR}/../windows/infill.ico)
    configure_file(../windows/launcher.rc.in ${CMAKE_BINARY_DIR}/launcher.rc @ONLY)
    add_executable(launcher WIN32 ../windows/launcher.c ${CMAKE_BINARY_DIR}/launcher.rc)
    set_target_properties(launcher PROPERTIES OUTPUT_NAME Infill)
    target_compile_definitions(launcher PRIVATE _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
    target_link_options(launcher PRIVATE -municode -static)
    target_link_libraries(launcher shell32)
else()
    target_link_libraries(infill_audio PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
